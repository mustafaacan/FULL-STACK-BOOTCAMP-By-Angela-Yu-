CREATE EXAMPLE

CREATE TABLE capitals (
    id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY, -- Auto increasement provided
    country VARCHAR(50) not null,
	capital VARCHAR(50) not null
);

** Primary = Unique + not null (we do not use these keywords)
** GENERATED ALWAYS AS IDENTITY --> acts like auto increasement, we do not need to add ID values while data inserting
